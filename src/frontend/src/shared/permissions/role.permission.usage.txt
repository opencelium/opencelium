role/permission

Page guard
export const ConnectorsPage = withPermission(
  'connector.read',
  { redirectTo: '/403' }
)(ConnectorsPageImpl)

Widget guard
export const ConnectorSettings = withPermission(
  'connector.update',
  { fallback: <readOnlyView /> }
)(ConnectorSettingsImpl)


Entity-aware permission
export const ConnectorEditor = withPermission(
  'connector.update',
  {
    getEntity: (props) => props.connector,
  }
)(Editor)

feature-level usage
if (!can({ permission: 'connector.delete', user, entity })) {
  notify.error('permission.denied')
  return
}
